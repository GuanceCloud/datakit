{{- if and .Values.gkeAutopilotPartner .Values.gkeAutopilotPartner.enabled }}
1. Check the Partner node agent:
  kubectl --namespace {{ .Release.Namespace }} rollout status daemonset/{{ include "datakit.fullname" . }}
  kubectl --namespace {{ .Release.Namespace }} get pods -l app.kubernetes.io/instance={{ .Release.Name }}

2. Check health through the Kubernetes Service proxy:
  kubectl get --raw "/api/v1/namespaces/{{ .Release.Namespace }}/services/http:{{ include "datakit.fullname" . }}:{{ .Values.service.port }}/proxy/v1/health"

The Partner allowlist disables exec, attach, and port-forward access.
{{- else }}
1. Get the application URL by running these commands:
  export POD_NAME=$(kubectl get pods --namespace {{ .Release.Namespace }} -l "app.kubernetes.io/name={{ include "datakit.name" . }},app.kubernetes.io/instance={{ .Release.Name }}" -o jsonpath="{.items[0].metadata.name}")
  export CONTAINER_PORT=$(kubectl get pod --namespace {{ .Release.Namespace }} $POD_NAME -o jsonpath="{.spec.containers[0].ports[0].containerPort}")
  echo "Visit http://127.0.0.1:9529 to use your application"
  kubectl --namespace {{ .Release.Namespace }} port-forward $POD_NAME 9529:$CONTAINER_PORT
{{- end }}
